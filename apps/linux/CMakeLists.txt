cmake_minimum_required(VERSION 3.21)
project(FileNodeLinux VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This frontend targets Linux. Use apps/windows for Windows WinUI.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2)
if(UNIX AND NOT APPLE)
  find_package(Qt6 REQUIRED COMPONENTS DBus)
endif()
qt_add_executable(filenode-ui main.cpp backend.cpp backend.h power.cpp power.h)
target_compile_definitions(filenode-ui PRIVATE FILENODE_VERSION="${PROJECT_VERSION}")
set_source_files_properties(../../locales/en.json PROPERTIES QT_RESOURCE_ALIAS "locales/en.json")
set_source_files_properties(../../locales/de.json PROPERTIES QT_RESOURCE_ALIAS "locales/de.json")
qt_add_resources(filenode-ui resources PREFIX "/" FILES Main.qml filenode.svg ../../locales/en.json ../../locales/de.json)
target_link_libraries(filenode-ui PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2)
if(UNIX AND NOT APPLE)
  target_link_libraries(filenode-ui PRIVATE Qt6::DBus)
endif()
install(TARGETS filenode-ui RUNTIME DESTINATION .)
