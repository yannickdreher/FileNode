cmake_minimum_required(VERSION 3.21)
project(FileNodeLinux VERSION 0.2.2 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This frontend targets Linux. Use apps/windows for Windows WinUI.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
option(BUILD_SHARED_LIBS "Build loadable native file-manager plugins" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2)
if(UNIX AND NOT APPLE)
  find_package(Qt6 REQUIRED COMPONENTS DBus)
endif()
add_subdirectory(kde)
add_subdirectory(integration)
