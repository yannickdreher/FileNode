find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(KF6 REQUIRED COMPONENTS CoreAddons KIO)
find_package(Qt6 REQUIRED COMPONENTS Concurrent Network)
kcoreaddons_add_plugin(filenodeoverlay SOURCES dolphin-overlay.cpp INSTALL_NAMESPACE "kf6/overlayicon")
target_link_libraries(filenodeoverlay PRIVATE KF6::KIOCore Qt6::Concurrent)
kcoreaddons_add_plugin(filenodeactions SOURCES dolphin-actions.cpp INSTALL_NAMESPACE "kf6/kfileitemaction")
target_link_libraries(filenodeactions PRIVATE KF6::KIOWidgets Qt6::Network)
set_target_properties(filenodeoverlay filenodeactions PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/integration")
add_executable(filenode-integration-smoke plugin-smoke.cpp)
target_link_libraries(filenode-integration-smoke PRIVATE KF6::KIOCore KF6::KIOWidgets KF6::CoreAddons Qt6::Widgets)
